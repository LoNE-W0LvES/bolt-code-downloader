'use client';

import { useEffect, useState } from 'react';
import { ContourEditorClient } from './ContourEditorClient';

interface Project {
  id: string;
  name: string;
  description?: string;
  directory_path: string;
  venv_status: 'uninitialized' | 'building' | 'ready' | 'failed';
  created_at: string;
  updated_at: string;
}

interface WorkspaceToggleProps {
  userRole: string;
}

export default function WorkspaceToggle({ userRole }: WorkspaceToggleProps) {
  const [subscribedProjects, setSubscribedProjects] = useState<Project[]>([]);
  const [selectedProject, setSelectedProject] = useState<Project | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Fetch user's subscribed projects (in this system, we fetch the projects list)
    fetch('/api/admin/projects')
      .then((res) => (res.ok ? res.json() : []))
      .then((data) => {
        setSubscribedProjects(data);
        setLoading(false);
      })
      .catch((err) => {
        console.error(err);
        setLoading(false);
      });
  }, []);

  if (loading) {
    return (
      <div className="flex-1 flex flex-col items-center justify-center text-center p-8 bg-background text-muted-foreground gap-4">
        <div className="w-10 h-10 border-4 border-primary border-t-transparent rounded-full animate-spin" />
        <div>
          <h2 className="font-headline font-bold text-foreground uppercase text-sm">
            Retrieving Subscribed Models
          </h2>
          <p className="text-[10px] mt-1">Loading model projects...</p>
        </div>
      </div>
    );
  }

  // If a project is selected, show the workspace/contour window for that specific ML model
  if (selectedProject) {
    return (
      <div className="flex-1 flex flex-col overflow-hidden bg-background">
        {/* Navigation / Header for Selected Project */}
        <div className="bg-card border-b border-border px-6 py-2 flex items-center justify-between shadow-sm select-none shrink-0">
          <div className="flex items-center gap-3">
            <button
              onClick={() => setSelectedProject(null)}
              className="px-3 py-1.5 rounded-md text-xs font-semibold border border-border bg-background hover:bg-muted transition-colors cursor-pointer"
            >
              ← Back to Models
            </button>
            <span className="text-xs font-bold text-foreground font-headline uppercase">
              📂 Model: {selectedProject.name}
            </span>
          </div>

          <div className="flex items-center gap-2 text-[10px] text-muted-foreground">
            <span>Active Role:</span>
            <span className="font-extrabold uppercase border border-border bg-background px-1.5 py-0.5 rounded text-foreground">
              {userRole}
            </span>
          </div>
        </div>

        <div className="flex-1 flex flex-col overflow-hidden">
          <ContourEditorClient />
        </div>
      </div>
    );
  }

  // Otherwise, show the list of subscribed ML models with thumbnail/cards
  return (
    <div className="flex-1 overflow-y-auto p-6 md:p-8 bg-background">
      <div className="max-w-6xl mx-auto flex flex-col gap-6">
        <div>
          <h1 className="font-headline text-2xl font-extrabold uppercase tracking-tight text-foreground">
            Subscribed ML Models
          </h1>
          <p className="text-muted-foreground text-xs mt-1 leading-relaxed">
            Select a machine learning model workspace to run inference, edit contour coordinates,
            and manage annotations.
          </p>
        </div>

        {subscribedProjects.length === 0 ? (
          <div className="border border-dashed border-border rounded-[var(--radius)] p-12 text-center flex flex-col items-center justify-center gap-3">
            <div className="w-12 h-12 rounded-full bg-muted flex items-center justify-center text-xl">
              🧠
            </div>
            <div>
              <h2 className="font-headline font-bold text-foreground uppercase text-xs">
                No Subscribed Models Available
              </h2>
              <p className="text-[10px] text-muted-foreground max-w-xs mt-1 leading-relaxed">
                You are not currently subscribed to any machine learning models. Please ask an
                administrator to set up a project workspace.
              </p>
            </div>
          </div>
        ) : (
          <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
            {subscribedProjects.map((project) => (
              <div
                key={project.id}
                onClick={() => setSelectedProject(project)}
                className="bg-card border border-border rounded-[var(--radius)] overflow-hidden shadow-sm hover:shadow-md hover:border-primary/50 transition-all cursor-pointer flex flex-col group"
              >
                {/* Thumbnail Header (Placeholder Visual Image representation) */}
                <div className="aspect-video bg-muted relative flex items-center justify-center overflow-hidden border-b border-border">
                  <div className="absolute inset-0 bg-[radial-gradient(var(--color-border)_1px,transparent_1px)] [background-size:16px_16px] opacity-40" />
                  <div className="relative border border-primary/20 w-3/4 h-3/5 bg-primary/5 rounded flex items-center justify-center transition-transform group-hover:scale-105">
                    <span className="text-[10px] text-primary font-mono font-bold uppercase tracking-wider">
                      {project.name} Workspace
                    </span>
                    <span className="absolute -top-1 -left-1 w-2 h-2 rounded-full bg-primary" />
                    <span className="absolute -bottom-1 -right-1 w-2 h-2 rounded-full bg-primary" />
                  </div>
                  <div className="absolute top-2 right-2">
                    <span
                      className={`text-[8px] px-1.5 py-0.5 rounded border uppercase tracking-wider font-extrabold ${
                        project.venv_status === 'ready'
                          ? 'border-emerald-500/20 text-emerald-600 bg-emerald-500/10'
                          : project.venv_status === 'building'
                            ? 'border-amber-500/20 text-amber-600 bg-amber-500/10'
                            : project.venv_status === 'failed'
                              ? 'border-destructive/20 text-destructive bg-destructive/10'
                              : 'border-border text-muted-foreground bg-muted'
                      }`}
                    >
                      {project.venv_status}
                    </span>
                  </div>
                </div>

                {/* Content Details */}
                <div className="p-4 flex flex-col gap-1.5">
                  <h3 className="font-bold text-sm text-foreground uppercase group-hover:text-primary transition-colors">
                    {project.name}
                  </h3>
                  <p className="text-muted-foreground text-xs line-clamp-2 leading-relaxed">
                    {project.description ||
                      'No description provided for this machine learning model.'}
                  </p>
                  <div className="flex items-center justify-between text-[10px] text-muted-foreground border-t border-border/50 pt-2.5 mt-1.5">
                    <span>Created: {new Date(project.created_at).toLocaleDateString()}</span>
                    <span className="font-semibold text-primary group-hover:underline">
                      Open Workspace →
                    </span>
                  </div>
                </div>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
